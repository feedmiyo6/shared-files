Exercise 1: Fixed-Priority Arbiter (Warmup)

Start with the simpler case. Implement a fixed-priority arbiter using the req & (~req + 1) trick.

module fixed_priority_arb #(
    parameter N = 4
)(
    input  logic [N-1:0] req,
    output logic [N-1:0] grant
);

    // Isolate the lowest set bit = highest priority requester
    assign grant = req & (~req + 1);

endmodule

💡 Why This Works

~req + 1 is the two's complement. When ANDed with req, all bits cancel except the lowest set bit. Example: req = 4'b1010, ~req+1 = 4'b0110, req & (~req+1) = 4'b0010 ✓

~req + 1은 2의 보수입니다. req와 AND하면 최하위 설정 비트를 제외한 모든 비트가 소거됩니다. 예: req = 4'b1010, ~req+1 = 4'b0110, req & (~req+1) = 4'b0010 ✓

Exercise 2: Round-Robin Arbiter

Now build the real deal: a mask-based round-robin arbiter with full fairness.

이제 진짜를 만들어 봅시다: 완전한 공정성을 갖춘 마스크 기반 라운드 로빈 아비터.

📋 Requirements

  • Parameterized N (default 4)
  • 파라미터화된 N (기본값 4)
  • Synchronous with active-low reset
  • 비동기 리셋 (active-low) 포함 동기식
  • Mask register tracks served masters
  • 마스크 레지스터로 서비스된 마스터 추적

💡 Hints

  1. Use the same x & (~x + 1) trick for both priority encoders
  2. 두 우선순위 인코더 모두에 x & (~x + 1) 트릭을 사용하세요
  3. After granting from masked: clear the granted bit AND all lower bits from mask
  4. 마스크에서 승인 후: 승인된 비트와 그 아래 모든 비트를 마스크에서 클리어
  5. After granting from raw (mask was empty): compute new mask excluding granted + below
  6. 원시에서 승인 후 (마스크가 비었을 때): 승인된 비트 + 아래를 제외한 새 마스크 계산
  7. Reset mask to all 1s
  8. 마스크를 모두 1로 리셋
module rr_arbiter #(
    parameter N = 4
)(
    input  logic              clk,
    input  logic              rst_n,
    input  logic [N-1:0]     req,
    output logic [N-1:0]     grant
);

    logic [N-1:0] mask;
    logic [N-1:0] masked_req, mask_grant, raw_grant;

    // Step 1: Apply mask to requests
    assign masked_req = req & mask;

    // Step 2: Priority encode both
    assign mask_grant = masked_req & (~masked_req + 1);
    assign raw_grant  = req & (~req + 1);

    // Step 3: Use masked grant if any masked request, else raw
    assign grant = (|masked_req) ? mask_grant : raw_grant;

    // Step 4: Update mask
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            mask <= {N{1'b1}};
        end else if (|grant) begin
            if (|masked_req)
                mask <= mask & ~((mask_grant << 1) - 1)
                         & ~mask_grant;
            else
                mask <= ~((raw_grant << 1) - 1)
                         & ~raw_grant;
        end
    end

endmodule

module tb_rr_arbiter;

    logic       clk, rst_n;
    logic [3:0] req, grant;
    int         count[4];

    rr_arbiter #(4) dut (.*);

    initial clk = 0;
    always #5 clk = ~clk;

    initial begin
        // Initialize
        for (int i = 0; i < 4; i++) count[i] = 0;
        req = 4'b0000;

        // Reset
        rst_n = 0;
        @(posedge clk);
        @(posedge clk);
        rst_n = 1;

        // All masters requesting for 12 cycles
        req = 4'b1111;
        repeat(12) begin
            @(posedge clk);
            #1;  // Small delay for grant to settle
            for (int i = 0; i < 4; i++)
                if (grant[i]) count[i]++;
            $display("req=%b grant=%b", req, grant);
        end

        // Verify fairness
        for (int i = 0; i < 4; i++) begin
            $display("Master %0d: %0d grants", i, count[i]);
            assert(count[i] == 3)
                else $error("Fairness violation! Master %0d got %0d grants",
                            i, count[i]);
        end

        $display("✅ All fairness checks passed!");
        $finish;
    end

endmodule

🌟 Bonus: Weighted Round-Robin

Extend your arbiter to support configurable weights per master. Each master gets weight[i] consecutive grants before the arbiter moves on.

아비터를 확장하여 마스터별 설정 가능한 가중치를 지원하세요. 각 마스터는 아비터가 다음으로 넘어가기 전에 weight[i]번의 연속 승인을 받습니다.

📋 Bonus Requirements

  • Add parameter int WEIGHT[N] = '{3, 1, 1, 1}
  • parameter int WEIGHT[N] = '{3, 1, 1, 1} 추가
  • Add a credit counter per master
  • 마스터별 크레딧 카운터 추가
  • Decrement credit on grant; skip master when credit is 0
  • 승인 시 크레딧 감소; 크레딧 0일 때 마스터 건너뜀
  • Reload all credits when a round completes
  • 라운드 완료 시 모든 크레딧 리로드
module wrr_arbiter #(
    parameter N = 4,
    parameter int WEIGHT[N] = '{3, 1, 1, 1}
)(
    input  logic                     clk,
    input  logic                     rst_n,
    input  logic [N-1:0]            req,
    output logic [N-1:0]            grant
);

    logic [$clog2(N)-1:0] ptr;
    int credit[N];

    always_comb begin
        grant = '0;
        // Search from ptr, wrapping around
        for (int i = 0; i < N; i++) begin
            automatic int idx = (ptr + i) % N;
            if (req[idx] && credit[idx] > 0) begin
                grant[idx] = 1'b1;
                break;
            end
        end
    end

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            ptr <= '0;
            for (int i = 0; i < N; i++)
                credit[i] <= WEIGHT[i];
        end else if (|grant) begin
            for (int i = 0; i < N; i++) begin
                if (grant[i]) begin
                    credit[i] <= credit[i] - 1;
                    // If credit exhausted, move pointer
                    if (credit[i] == 1)
                        ptr <= (i + 1) % N;
                end
            end

            // Reload credits when all exhausted
            if (credit[0] + credit[1] +
                credit[2] + credit[3] <= 1) begin
                for (int i = 0; i < N; i++)
                    credit[i] <= WEIGHT[i];
            end
        end
    end

endmodule

🏆 Interview Challenge

Common Interview Question

"Design a round-robin arbiter for N requesters. How would you handle the case where a master holds its request for multiple cycles?"

"N개의 요청자를 위한 라운드 로빈 아비터를 설계하세요. 마스터가 여러 사이클 동안 요청을 유지하는 경우는 어떻게 처리하시겠습니까?"

Think about: Should the arbiter only rotate when the current master releases its request? Or should it forcefully rotate every cycle? This leads to the concept of preemptive vs non-preemptive arbitration.

생각해볼 점: 아비터는 현재 마스터가 요청을 해제할 때만 회전해야 할까요? 아니면 매 사이클마다 강제 회전해야 할까요? 이는 선점형(preemptive) vs 비선점형(non-preemptive) 중재 개념으로 이어집니다.
