💡 Hint: Use casez on addr[31:28]. Remember that each target_sel bit is one-hot — only one should be high at a time. addr[31:28]에 casez를 사용하세요. 각 target_sel 비트는 원핫입니다 — 한 번에 하나만 high여야 합니다. 
module addr_decoder (
    input  logic [31:0] addr,
    input  logic        valid,
    output logic [1:0]  target_sel,
    output logic        addr_error
);

    always_comb begin
        target_sel = 2'b00;
        addr_error = 1'b0;

        if (valid) begin
            casez (addr[31:28])
                4'h0:    target_sel[0] = 1'b1;  // 0x0xxx → Target 0
                4'h4:    target_sel[1] = 1'b1;  // 0x4xxx → Target 1
                default: addr_error    = 1'b1;  // Unmapped!
            endcase
        end
    end

endmodule

module arb_mux (
    input  logic        i0_valid,
    output logic        i0_ready,
    input  logic [31:0] i0_addr,
    input  logic [31:0] i0_data,
    input  logic        i1_valid,
    output logic        i1_ready,
    input  logic [31:0] i1_addr,
    input  logic [31:0] i1_data,
    output logic        t_valid,
    input  logic        t_ready,
    output logic [31:0] t_addr,
    output logic [31:0] t_data,
    output logic        t_id
);

    always_comb begin
        // Default: nothing granted
        i0_ready = 1'b0;
        i1_ready = 1'b0;
        t_valid  = 1'b0;
        t_addr   = 32'h0;
        t_data   = 32'h0;
        t_id     = 1'b0;

        if (i0_valid) begin
            // Initiator 0 wins (higher priority)
            t_valid  = 1'b1;
            t_addr   = i0_addr;
            t_data   = i0_data;
            t_id     = 1'b0;
            i0_ready = t_ready;  // Pass back ready
        end else if (i1_valid) begin
            // Initiator 1 gets access when 0 is idle
            t_valid  = 1'b1;
            t_addr   = i1_addr;
            t_data   = i1_data;
            t_id     = 1'b1;
            i1_ready = t_ready;  // Pass back ready
        end
    end

endmodule

module simple_bus_top (
    input  logic        clk, rst_n,
    input  logic        i0_valid,
    output logic        i0_ready,
    input  logic [31:0] i0_addr,
    input  logic [31:0] i0_wdata,
    input  logic        i0_wr,
    input  logic        i1_valid,
    output logic        i1_ready,
    input  logic [31:0] i1_addr,
    input  logic [31:0] i1_wdata,
    input  logic        i1_wr,
    output logic [31:0] tgt0_ctrl, tgt0_cfg,
    output logic [31:0] tgt1_ctrl, tgt1_cfg
);

    // Internal wires: arbiter → targets
    logic        arb_valid, arb_ready;
    logic [31:0] arb_addr, arb_data;
    logic        arb_id;

    // Decoder outputs
    logic [1:0]  target_sel;
    logic        addr_error;

    // Target ready signals
    logic        t0_ready, t1_ready;

    // 1. Arbitrate between initiators
    arb_mux u_arb (
        .i0_valid(i0_valid), .i0_ready(i0_ready),
        .i0_addr(i0_addr),   .i0_data(i0_wdata),
        .i1_valid(i1_valid), .i1_ready(i1_ready),
        .i1_addr(i1_addr),   .i1_data(i1_wdata),
        .t_valid(arb_valid), .t_ready(arb_ready),
        .t_addr(arb_addr),   .t_data(arb_data),
        .t_id(arb_id)
    );

    // 2. Decode address
    addr_decoder u_dec (
        .addr(arb_addr),
        .valid(arb_valid),
        .target_sel(target_sel),
        .addr_error(addr_error)
    );

    // 3. Target 0 — register block
    reg_block u_tgt0 (
        .clk(clk), .rst_n(rst_n),
        .req_valid(arb_valid & target_sel[0]),
        .req_ready(t0_ready),
        .req_addr(arb_addr[7:0]),
        .req_wdata(arb_data),
        .req_wr(i0_wr),  // Simplified: use wr from arb
        .resp_valid(), .resp_ready(1'b1),
        .resp_rdata(),
        .ctrl_reg(tgt0_ctrl), .cfg_reg(tgt0_cfg),
        .status_reg(32'hAAAA_BBBB)
    );

    // 4. Target 1 — register block
    reg_block u_tgt1 (
        .clk(clk), .rst_n(rst_n),
        .req_valid(arb_valid & target_sel[1]),
        .req_ready(t1_ready),
        .req_addr(arb_addr[7:0]),
        .req_wdata(arb_data),
        .req_wr(i1_wr),
        .resp_valid(), .resp_ready(1'b1),
        .resp_rdata(),
        .ctrl_reg(tgt1_ctrl), .cfg_reg(tgt1_cfg),
        .status_reg(32'hCCCC_DDDD)
    );

    // 5. Collect ready from selected target (or error)
    assign arb_ready = (target_sel[0] & t0_ready)
                     | (target_sel[1] & t1_ready)
                     | addr_error;  // Error → ack immediately

endmodule
