Exercise 1: Pipeline Stage with Stall

🎯 Goal

Build a single pipeline stage that supports stalling. When stall = 1, the stage holds its data. When stall = 0, data passes through.

스톨을 지원하는 단일 파이프라인 스테이지를 만드세요. stall = 1이면 데이터를 유지하고, stall = 0이면 데이터가 통과합니다.

    output logic             valid_out

    // When stall=1: hold current values
    // When stall=0: capture data_in and valid_in

        @(posedge clk);
        data_in = 8'hBB;
        @(posedge clk);
        $display("[Normal] data_out=%h valid=%b (expect BB, 1)", data_out, valid_out);

        // Test 2: Stall — data should freeze
        stall = 1;
        data_in = 8'hCC;
        @(posedge clk);
        @(posedge clk);
        $display("[Stall]  data_out=%h valid=%b (expect BB, 1)", data_out, valid_out);

        // Test 3: Release stall
        stall = 0;
        @(posedge clk);
        $display("[Resume] data_out=%h valid=%b (expect CC, 1)", data_out, valid_out);

        // Test 4: Valid goes low
        valid_in = 0; data_in = 8'hFF;
        @(posedge clk);
        $display("[NoData] data_out=%h valid=%b (expect FF, 0)", data_out, valid_out);

        #20 $finish;
    end
endmodule
module pipe_stage #(
    parameter WIDTH = 8
)(
    input  logic             clk,
    input  logic             rst_n,
    input  logic             stall,
    input  logic [WIDTH-1:0] data_in,
    input  logic             valid_in,
    output logic [WIDTH-1:0] data_out,
    output logic             valid_out
);

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            data_out  <= '0;
            valid_out <= 1'b0;
        end else if (!stall) begin
            data_out  <= data_in;
            valid_out <= valid_in;
        end
        // Implicit: when stalled, hold values
    end

endmodule

Exercise 2: 3-Stage Pipeline with Bubble Insertion

🎯 Goal

Build a 3-stage pipeline (S1→S2→S3) with a controller that can:
• Stall all stages when downstream_busy is asserted
• Insert a bubble into S2 when hazard is detected (stall S1, bubble S2, S3 keeps going)

3단계 파이프라인(S1→S2→S3)과 컨트롤러를 구축하세요:
• downstream_busy시 모든 스테이지를 스톨
• hazard 감지 시 S2에 버블 삽입 (S1 스톨, S2 버블, S3 계속 진행)

💡 Hint

Remember: squash_s2 = hazard & !downstream_busy — never squash during a stall! If downstream_busy is high, S2 is stalled and squashing would destroy valid data.

squash_s2 = hazard & !downstream_busy — 스톨 중에는 절대 스쿼시하지 마세요! downstream_busy가 높으면 S2가 스톨되어 스쿼시하면 유효한 데이터가 파괴됩니다.

module pipe_stage_squash #(
    parameter WIDTH = 8
)(
    input  logic             clk,
    input  logic             rst_n,
    input  logic             stall,
    input  logic             squash,
    input  logic [WIDTH-1:0] data_in,
    input  logic             valid_in,
    output logic [WIDTH-1:0] data_out,
    output logic             valid_out
);

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            data_out  <= '0;
            valid_out <= 1'b0;
        end else if (squash) begin
            valid_out <= 1'b0;  // Insert bubble
        end else if (!stall) begin
            data_out  <= data_in;
            valid_out <= valid_in;
        end
    end

endmodule

module pipeline_3stage (
    input  logic       clk,
    input  logic       rst_n,
    input  logic [7:0] data_in,
    input  logic       valid_in,
    input  logic       hazard,
    input  logic       downstream_busy,
    output logic [7:0] data_out,
    output logic       valid_out,
    output logic       stall_upstream
);

    logic [7:0] s1_data, s2_data;
    logic       s1_valid, s2_valid;
    logic       stall_s1, stall_s2, stall_s3;
    logic       squash_s2;

    // Control signals
    assign stall_s3   = downstream_busy;
    assign stall_s2   = downstream_busy;
    assign stall_s1   = downstream_busy | hazard;
    assign squash_s2  = hazard & !downstream_busy;
    assign stall_upstream = stall_s1;

    // Stage 1: simple stall
    pipe_stage s1 (
        .clk(clk), .rst_n(rst_n),
        .stall(stall_s1),
        .data_in(data_in), .valid_in(valid_in),
        .data_out(s1_data), .valid_out(s1_valid)
    );

    // Stage 2: stall + squash (bubble insertion)
    pipe_stage_squash s2 (
        .clk(clk), .rst_n(rst_n),
        .stall(stall_s2),
        .squash(squash_s2),
        .data_in(s1_data), .valid_in(s1_valid),
        .data_out(s2_data), .valid_out(s2_valid)
    );

    // Stage 3: simple stall
    pipe_stage s3 (
        .clk(clk), .rst_n(rst_n),
        .stall(stall_s3),
        .data_in(s2_data), .valid_in(s2_valid),
        .data_out(data_out), .valid_out(valid_out)
    );

endmodule

Exercise 3: Skid Buffer

🎯 Goal

Implement a skid buffer with ready/valid interfaces on both sides. The buffer should:
• Pass data through directly when downstream is ready (zero latency)
• Capture one entry when downstream stalls (absorb 1 cycle)
• Deassert up_ready when the internal buffer is full

양쪽에 ready/valid 인터페이스가 있는 스키드 버퍼를 구현하세요:
• 다운스트림이 준비되면 데이터를 직접 전달 (제로 레이턴시)
• 다운스트림이 스톨하면 한 엔트리 캡처 (1사이클 흡수)
• 내부 버퍼가 차면 up_ready 디어서트

Starter Code (design.sv)

module skid_buffer #(
    parameter WIDTH = 8
)(
    input  logic             clk,
    input  logic             rst_n,
    // Upstream
    input  logic [WIDTH-1:0] up_data,
    input  logic             up_valid,
    output logic             up_ready,
    // Downstream
    output logic [WIDTH-1:0] dn_data,
    output logic             dn_valid,
    input  logic             dn_ready
);

    logic [WIDTH-1:0] buf_data;
    logic             buf_valid;

    // TODO: Implement
    // 1. up_ready = !buf_valid (accept when buffer empty)
    // 2. dn_data = buf_valid ? buf_data : up_data
    // 3. dn_valid = buf_valid | up_valid
    // 4. Sequential logic:
    //    - Reset: buf_valid = 0
    //    - Buffer full + dn_ready: drain buffer (buf_valid=0)
    //    - up_valid + up_ready + !dn_ready: fill buffer

endmodule

Testbench (testbench.sv)

module tb_skid_buffer;
    logic       clk, rst_n;
    logic [7:0] up_data, dn_data;
    logic       up_valid, up_ready, dn_valid, dn_ready;

    skid_buffer dut (.*);

    initial clk = 0;
    always #5 clk = ~clk;

    initial begin
        $dumpfile("dump.vcd"); $dumpvars;

        rst_n = 0; up_valid = 0; up_data = 0; dn_ready = 1;
        #20 rst_n = 1;

        // Test 1: Passthrough (dn_ready=1)
        @(posedge clk);
        up_data = 8'h11; up_valid = 1;
        #1;
        $display("[Pass] dn_data=%h dn_valid=%b up_ready=%b (expect 11,1,1)",
                 dn_data, dn_valid, up_ready);

        // Test 2: Downstream stalls — data captured into buffer
        @(posedge clk);
        dn_ready = 0;
        up_data = 8'h22; up_valid = 1;
        @(posedge clk);
        #1;
        $display("[Stall] dn_data=%h buf up_ready=%b (expect 22, ready=0)",
                 dn_data, up_ready);

        // Test 3: Downstream resumes — buffer drains
        up_valid = 0;
        dn_ready = 1;
        @(posedge clk);
        #1;
        $display("[Drain] dn_data=%h up_ready=%b (expect 22->drain, ready=1)",
                 dn_data, up_ready);

        #40 $finish;
    end
endmodule
module skid_buffer #(
    parameter WIDTH = 8
)(
    input  logic             clk,
    input  logic             rst_n,
    input  logic [WIDTH-1:0] up_data,
    input  logic             up_valid,
    output logic             up_ready,
    output logic [WIDTH-1:0] dn_data,
    output logic             dn_valid,
    input  logic             dn_ready
);

    logic [WIDTH-1:0] buf_data;
    logic             buf_valid;

    // Combinational outputs
    assign up_ready = !buf_valid;
    assign dn_data  = buf_valid ? buf_data : up_data;
    assign dn_valid = buf_valid | up_valid;

    // Buffer management
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            buf_valid <= 1'b0;
            buf_data  <= '0;
        end else if (buf_valid && dn_ready) begin
            // Drain: downstream consumed buffered data
            buf_valid <= 1'b0;
        end else if (up_valid && up_ready && !dn_ready) begin
            // Fill: upstream sending but downstream busy
            buf_data  <= up_data;
            buf_valid <= 1'b1;
        end
    end

endmodule

🏆 Bonus Challenge: Throughput Counter

🎯 Advanced Goal

Add performance counters to your 3-stage pipeline that track:
• total_cycles: total clock cycles elapsed
• valid_cycles: cycles where valid data exits the pipeline
• stall_cycles: cycles where the pipeline is stalled
• bubble_cycles: cycles where a bubble flows through
Then calculate effective throughput at the end of simulation!

3단계 파이프라인에 성능 카운터를 추가하세요:
• total_cycles: 경과한 총 클럭 사이클
• valid_cycles: 유효한 데이터가 파이프라인을 나가는 사이클
• stall_cycles: 파이프라인이 스톨된 사이클
• bubble_cycles: 버블이 통과하는 사이클
시뮬레이션 끝에 유효 처리량을 계산하세요!

// At end of simulation:
initial begin
    // ... your test sequence ...
