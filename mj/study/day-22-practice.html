Exercise 1: Pipeline a 2-Operation Datapath

Take this unpipelined module and convert it to a 2-stage pipeline. Stage 1 does the multiply, Stage 2 does the XOR. Don't forget pass-through registers!

이 파이프라인 없는 모듈을 2단계 파이프라인으로 변환하세요. Stage 1은 곱셈, Stage 2는 XOR을 수행합니다. 패스스루 레지스터를 잊지 마세요!

// UNPIPELINED — convert this to 2-stage pipeline
module compute (
    input  logic        clk, rst_n,
    input  logic [7:0]  a, b,
    input  logic [15:0] mask,
    output logic [15:0] result
);
    always_ff @(posedge clk or negedge rst_n)
        if (!rst_n) result <= '0;
        else        result <= (a * b) ^ mask;
endmodule
Hints:
  • Stage 1: compute a * b, store in s1_product
  • Stage 1: a * b를 계산하여 s1_product에 저장
  • Which signal needs a pass-through register? mask — it's used in Stage 2!
  • 어떤 신호에 패스스루 레지스터가 필요한가요? mask — Stage 2에서 사용됩니다!
📖 Show Solution 
module compute_pipelined (
    input  logic        clk, rst_n,
    input  logic [7:0]  a, b,
    input  logic [15:0] mask,
    output logic [15:0] result
);

    // Stage 1 registers
    logic [15:0] s1_product;
    logic [15:0] s1_mask;    // pass-through!

    always_ff @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            s1_product <= '0;
            s1_mask    <= '0;
        end else begin
            s1_product <= a * b;
            s1_mask    <= mask;
        end

    // Stage 2
    always_ff @(posedge clk or negedge rst_n)
        if (!rst_n) result <= '0;
        else        result <= s1_product ^ s1_mask;

endmodule

module compute_pipelined_v (
    input  logic        clk, rst_n,
    input  logic        input_valid,
    input  logic [7:0]  a, b,
    input  logic [15:0] mask,
    output logic [15:0] result,
    output logic        output_valid
);

    // Stage 1 registers
    logic [15:0] s1_product;
    logic [15:0] s1_mask;
    logic        s1_valid;

    always_ff @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            s1_product <= '0;
            s1_mask    <= '0;
            s1_valid   <= 1'b0;
        end else begin
            s1_product <= a * b;
            s1_mask    <= mask;
            s1_valid   <= input_valid;
        end

    // Stage 2
    always_ff @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            result       <= '0;
            output_valid <= 1'b0;
        end else begin
            result       <= s1_product ^ s1_mask;
            output_valid <= s1_valid;
        end

endmodule

Analysis

Option A: Cut after Multiply → S1: 8ns, S2: 14ns → fmax = 71 MHz

🏆 Challenge: 3-Stage Pipeline with Valid

Design a 3-stage pipelined ALU that performs: result = ((a + b) * c) >> shift_amt. Include valid signals and proper pass-through registers. Use the s1_/s2_ naming convention.

다음을 수행하는 3단계 파이프라인 ALU를 설계하세요: result = ((a + b) * c) >> shift_amt. valid 신호와 적절한 패스스루 레지스터를 포함하세요. s1_/s2_ 명명 규칙을 사용하세요.

Stage Plan:
  • Stage 1: Add (a + b) — pass-through: c, shift_amt
  • Stage 1: 덧셈 (a + b) — 패스스루: c, shift_amt
  • Stage 2: Multiply (sum * c) — pass-through: shift_amt
  • Stage 2: 곱셈 (sum * c) — 패스스루: shift_amt
  • Stage 3: Shift (product >> shift_amt)
  • Stage 3: 시프트 (product >> shift_amt)
📖 Show Solution 
module pipe3_alu (
    input  logic        clk, rst_n,
    input  logic        in_valid,
    input  logic [7:0]  a, b, c,
    input  logic [2:0]  shift_amt,
    output logic [15:0] result,
    output logic        out_valid
);

    // ===== Stage 1: Add =====
    logic [8:0]  s1_sum;
    logic [7:0]  s1_c;           // pass-through
    logic [2:0]  s1_shift_amt;   // pass-through
    logic        s1_valid;

    always_ff @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            s1_sum       <= '0;
            s1_c         <= '0;
            s1_shift_amt <= '0;
            s1_valid     <= 1'b0;
        end else begin
            s1_sum       <= a + b;
            s1_c         <= c;
            s1_shift_amt <= shift_amt;
            s1_valid     <= in_valid;
        end

    // ===== Stage 2: Multiply =====
    logic [16:0] s2_product;
    logic [2:0]  s2_shift_amt;  // pass-through
    logic        s2_valid;

    always_ff @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            s2_product   <= '0;
            s2_shift_amt <= '0;
            s2_valid     <= 1'b0;
        end else begin
            s2_product   <= s1_sum * s1_c;
            s2_shift_amt <= s1_shift_amt;
            s2_valid     <= s1_valid;
        end

    // ===== Stage 3: Shift =====
    always_ff @(posedge clk or negedge rst_n)
        if (!rst_n) begin
            result    <= '0;
            out_valid <= 1'b0;
        end else begin
            result    <= s2_product[15:0] >> s2_shift_amt;
            out_valid <= s2_valid;
        end

endmodule
