EASY Exercise 1: Basic Ready/Valid Source 연습 1: 기본 Ready/Valid 소스

Implement a simple source that sends values 0, 1, 2, 3, ... using the ready/valid protocol. The source increments its counter only after each successful transfer.

ready/valid 프로토콜을 사용하여 0, 1, 2, 3, ...을 보내는 간단한 소스를 구현하세요. 소스는 각 성공적인 전송 후에만 카운터를 증가시킵니다.

module counting_source #(
    parameter WIDTH = 8
)(
    input  logic             clk,
    input  logic             rst_n,
    output logic             out_valid,
    input  logic             out_ready,
    output logic [WIDTH-1:0] out_data
);

    logic [WIDTH-1:0] counter;

    // TODO: Implement the counting source
    // 1. out_valid should always be high (we always have data)
    // 2. out_data should output the current counter value
    // 3. Increment counter ONLY on successful transfer
    //    (when both valid AND ready are high)

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            // TODO: Reset counter
        end else begin
            // TODO: Increment on transfer
        end
    end

    // TODO: Drive out_valid and out_data

endmodule

// ─── Testbench ───
module tb_counting_source;
    logic clk, rst_n;
    logic out_valid, out_ready;
    logic [7:0] out_data;

    counting_source dut (.*);

    initial clk = 0;
    always #5 clk = ~clk;

    initial begin
        rst_n = 0; out_ready = 0;
        #20 rst_n = 1;

        // Test 1: Ready always high — should count 0,1,2,3...
        out_ready = 1;
        repeat(4) @(posedge clk);

        // Test 2: Deassert ready — counter should STALL
        out_ready = 0;
        repeat(3) @(posedge clk);

        // Test 3: Reassert ready — should continue from where it stopped
        out_ready = 1;
        repeat(4) @(posedge clk);

        $display("Test complete!");
        $finish;
    end

    always @(posedge clk) begin
        if (rst_n)
            $display("[%0t] valid=%b ready=%b data=%0d %s",
                $time, out_valid, out_ready, out_data,
                (out_valid & out_ready) ? "<-- TRANSFER" : "");
    end
endmodule

🔑 Hint: The transfer condition is simply out_valid & out_ready. Since this source always has data, out_valid can be driven as a constant 1'b1 after reset.

🔑 힌트: 전송 조건은 간단히 out_valid & out_ready입니다. 이 소스는 항상 데이터가 있으므로 리셋 후 out_valid를 상수 1'b1로 구동할 수 있습니다.

📋 Solution📋 솔루션 
assign out_valid = rst_n;  // Always valid after reset
assign out_data  = counter;

always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        counter <= '0;
    end else if (out_valid & out_ready) begin
        counter <= counter + 1;
    end
end

MEDIUM Exercise 2: Ready/Valid Pipeline Stage 연습 2: Ready/Valid 파이프라인 스테이지

Build a pipeline stage that doubles its input data. It accepts data via ready/valid on the input side and presents the doubled result via ready/valid on the output side. This stage must properly propagate backpressure.

입력 데이터를 두 배로 만드는 파이프라인 스테이지를 구현하세요. 입력 측에서 ready/valid로 데이터를 받고 출력 측에서 ready/valid로 두 배 결과를 표시합니다. 이 스테이지는 백프레셔를 올바르게 전파해야 합니다.

module double_stage #(
    parameter WIDTH = 8
)(
    input  logic             clk,
    input  logic             rst_n,

    // Input interface (we are the SINK)
    input  logic             in_valid,
    output logic             in_ready,
    input  logic [WIDTH-1:0] in_data,

    // Output interface (we are the SOURCE)
    output logic             out_valid,
    input  logic             out_ready,
    output logic [WIDTH-1:0] out_data
);

    // TODO: Implement the pipeline stage
    // Requirements:
    // 1. in_ready = out_ready | !out_valid
    // 2. When in_ready, capture in_data, double it, store in out_data
    // 3. out_valid tracks whether we have valid data stored
    // 4. The "doubling" is: out_data = in_data << 1 (left shift by 1)

endmodule

// ─── Testbench ───
module tb_double_stage;
    logic clk, rst_n;
    logic in_valid, in_ready;
    logic [7:0] in_data;
    logic out_valid, out_ready;
    logic [7:0] out_data;

    double_stage dut (.*);

    initial clk = 0;
    always #5 clk = ~clk;

    task send(input [7:0] val);
        in_valid = 1; in_data = val;
        @(posedge clk);
        while (!in_ready) @(posedge clk);
        in_valid = 0;
    endtask

    initial begin
        rst_n = 0; in_valid = 0; out_ready = 1; in_data = 0;
        #20 rst_n = 1;

        // Send values and check doubled output
        send(5);   // expect out_data = 10
        send(10);  // expect out_data = 20
        send(42);  // expect out_data = 84

        // Test backpressure: stall downstream
        out_ready = 0;
        send(7);
        #30;
        $display("Stalled: out_valid=%b out_data=%0d", out_valid, out_data);
        out_ready = 1;
        #20;

        $display("Test complete!");
        $finish;
    end

    always @(posedge clk) begin
        if (rst_n && out_valid && out_ready)
            $display("[%0t] OUTPUT: %0d (doubled)", $time, out_data);
    end
endmodule

🔑 Hint: The key expression is in_ready = out_ready | !out_valid. When in_ready is true and in_valid is true, latch the input, double it (in_data << 1), and set out_valid.

🔑 힌트: 핵심 표현식은 in_ready = out_ready | !out_valid입니다. in_ready가 true이고 in_valid가 true이면, 입력을 래치하고 두 배로 만들어 (in_data << 1) out_valid를 설정하세요.

📋 Solution📋 솔루션 
assign in_ready = out_ready | !out_valid;

always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        out_valid <= 1'b0;
        out_data  <= '0;
    end else if (in_ready) begin
        out_valid <= in_valid;
        out_data  <= in_data << 1;  // Double!
    end
end

HARD Exercise 3: Skid Buffer Implementation 연습 3: 스키드 버퍼 구현

Implement a full skid buffer. This is the most important flow-control building block in real SoC design. The key insight: in_ready must NOT depend combinationally on out_ready.

완전한 스키드 버퍼를 구현하세요. 이것은 실제 SoC 설계에서 가장 중요한 흐름 제어 빌딩 블록입니다. 핵심 통찰: in_ready는 조합적으로 out_ready에 의존하면 안 됩니다.

module skid_buffer #(
    parameter WIDTH = 8
)(
    input  logic             clk,
    input  logic             rst_n,

    // Input interface
    input  logic             in_valid,
    output logic             in_ready,
    input  logic [WIDTH-1:0] in_data,

    // Output interface
    output logic             out_valid,
    input  logic             out_ready,
    output logic [WIDTH-1:0] out_data
);

    // Two registers: main (output) and skid (overflow)
    logic [WIDTH-1:0] main_data, skid_data;
    logic main_valid, skid_valid;

    // TODO: Implement the skid buffer
    // Key requirements:
    // 1. in_ready = !skid_valid (no comb path to out_ready!)
    // 2. out_valid = main_valid
    // 3. out_data = main_data
    // 4. When downstream accepts (out_ready | !main_valid):
    //    - If skid has data → move skid to main
    //    - Else if input has data → move input to main
    //    - Else → main_valid = 0
    // 5. When downstream stalled but input has data:
    //    - Catch input in skid register

endmodule

// ─── Testbench ───
module tb_skid_buffer;
    logic clk, rst_n;
    logic in_valid, in_ready;
    logic [7:0] in_data;
    logic out_valid, out_ready;
    logic [7:0] out_data;
    int errors = 0;

    skid_buffer dut (.*);

    initial clk = 0;
    always #5 clk = ~clk;

    task automatic check_output(input [7:0] expected);
        if (out_valid && out_ready) begin
            if (out_data !== expected) begin
                $display("ERROR: got %0d, expected %0d", out_data, expected);
                errors++;
            end else
                $display("OK: transferred %0d", out_data);
        end
    endtask

    initial begin
        rst_n = 0; in_valid = 0; out_ready = 0; in_data = 0;
        #20 rst_n = 1;

        // Test 1: Continuous flow (both ready)
        $display("\n--- Test 1: Continuous flow ---");
        out_ready = 1;
        for (int i = 0; i < 5; i++) begin
            in_valid = 1; in_data = i * 10;
            @(posedge clk);
        end
        in_valid = 0;
        repeat(3) @(posedge clk);

        // Test 2: Downstream stalls then resumes
        $display("\n--- Test 2: Backpressure test ---");
        out_ready = 0;
        in_valid = 1; in_data = 8'hAA;
        @(posedge clk);
        in_data = 8'hBB;  // This should go to skid
        @(posedge clk);
        in_valid = 0;
        repeat(2) @(posedge clk);

        $display("Stalled state: out_valid=%b in_ready=%b", out_valid, in_ready);

        // Release backpressure
        out_ready = 1;
        repeat(5) @(posedge clk);

        if (errors == 0) $display("\n✅ All tests passed!");
        else $display("\n❌ %0d errors found", errors);
        $finish;
    end

    always @(posedge clk) begin
        if (rst_n)
            $display("[%0t] in: v=%b r=%b d=%02h | out: v=%b r=%b d=%02h",
                $time, in_valid, in_ready, in_data,
                out_valid, out_ready, out_data);
    end
endmodule

🔑 Hint: Think of two states: (1) downstream is accepting or main is empty → pull from skid first, then input. (2) downstream is stalled AND main is full → catch incoming data in the skid register. The critical insight: in_ready = !skid_valid — no combinational path from out_ready!

🔑 힌트: 두 가지 상태를 생각하세요: (1) 다운스트림이 수락하거나 메인이 비어있으면 → 스키드에서 먼저, 그다음 입력에서 가져옴. (2) 다운스트림이 멈추고 메인이 가득 차면 → 들어오는 데이터를 스키드 레지스터에 잡음. 핵심 통찰: in_ready = !skid_valid — out_ready에서 조합 경로 없음!

📋 Solution📋 솔루션 
assign in_ready  = !skid_valid;
assign out_valid = main_valid;
assign out_data  = main_data;

always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        main_valid <= 1'b0;
        skid_valid <= 1'b0;
        main_data  <= '0;
        skid_data  <= '0;
    end else begin
        if (out_ready | !main_valid) begin
            if (skid_valid) begin
                main_data  <= skid_data;
                main_valid <= 1'b1;
                skid_valid <= 1'b0;
            end else if (in_valid && in_ready) begin
                main_data  <= in_data;
                main_valid <= 1'b1;
            end else begin
                main_valid <= 1'b0;
            end
        end else if (in_valid && in_ready) begin
            skid_data  <= in_data;
            skid_valid <= 1'b1;
        end
    end
end

HARD Exercise 4: Ready/Valid Data Aggregator 연습 4: Ready/Valid 데이터 집계기

Design a module that accepts 4 individual bytes (one per cycle via ready/valid) and outputs a 32-bit word. This is a common pattern in network protocol processing — assembling packet headers from byte streams.

4개의 개별 바이트를 받아(ready/valid를 통해 사이클당 하나) 32비트 워드를 출력하는 모듈을 설계하세요. 이것은 네트워크 프로토콜 처리에서 일반적인 패턴입니다 — 바이트 스트림에서 패킷 헤더를 조립합니다.

module byte_to_word (
    input  logic        clk,
    input  logic        rst_n,

    // Byte input (8-bit)
    input  logic        in_valid,
    output logic        in_ready,
    input  logic [7:0]  in_data,

    // Word output (32-bit)
    output logic        out_valid,
    input  logic        out_ready,
    output logic [31:0] out_data
);

    // TODO: Implement byte-to-word aggregator
    // Requirements:
    // 1. Accept 4 bytes sequentially via ready/valid input
    // 2. First byte → bits [31:24], second → [23:16], etc.
    // 3. After 4 bytes, assert out_valid with 32-bit word
    // 4. Stop accepting (in_ready=0) until output is taken
    // 5. After output transfer, start collecting next 4 bytes
    //
    // Hint: Use a 2-bit counter to track which byte you're on
    // Hint: Use a shift register to accumulate bytes

endmodule

// ─── Testbench ───
module tb_byte_to_word;
    logic clk, rst_n;
    logic in_valid, in_ready;
    logic [7:0] in_data;
    logic out_valid, out_ready;
    logic [31:0] out_data;

    byte_to_word dut (.*);

    initial clk = 0;
    always #5 clk = ~clk;

    initial begin
        rst_n = 0; in_valid = 0; out_ready = 1; in_data = 0;
        #20 rst_n = 1;

        // Send bytes: 0xDE, 0xAD, 0xBE, 0xEF
        // Expected output word: 0xDEADBEEF
        in_valid = 1;
        in_data = 8'hDE; @(posedge clk);
        in_data = 8'hAD; @(posedge clk);
        in_data = 8'hBE; @(posedge clk);
        in_data = 8'hEF; @(posedge clk);
        in_valid = 0;

        repeat(3) @(posedge clk);

        $display("Output: 0x%08h (expected 0xDEADBEEF)", out_data);
        $finish;
    end

    always @(posedge clk) begin
        if (out_valid && out_ready)
            $display("[%0t] WORD OUTPUT: 0x%08h", $time, out_data);
    end
endmodule
📋 Solution📋 솔루션 
logic [1:0]  byte_cnt;
logic [31:0] shift_reg;
logic        word_ready;

assign word_ready = (byte_cnt == 2'd3) && in_valid && in_ready;
assign in_ready   = !out_valid;  // Accept bytes when no word pending

always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        byte_cnt  <= 2'd0;
        shift_reg <= 32'd0;
        out_valid <= 1'b0;
        out_data  <= 32'd0;
    end else begin
        // Clear output when downstream accepts
        if (out_valid && out_ready)
            out_valid <= 1'b0;

        // Accept input bytes
        if (in_valid && in_ready) begin
            shift_reg <= {shift_reg[23:0], in_data};
            if (byte_cnt == 2'd3) begin
                out_data  <= {shift_reg[23:0], in_data};
                out_valid <= 1'b1;
                byte_cnt  <= 2'd0;
            end else begin
                byte_cnt  <= byte_cnt + 1;
            end
        end
    end
end
