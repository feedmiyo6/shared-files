module data_processor (
    input  logic clk,
    input  logic rst_n,      // FIX 1: Added reset
    input  logic req,
    input  logic data_ready,
    output logic ack,
    output logic busy,
    output logic error      // FIX 3: Error output
);
    typedef enum logic [1:0] {
        IDLE      = 2'b00,
        WAIT_DATA = 2'b01,
        PROCESS   = 2'b10,
        RESPOND   = 2'b11
    } state_t;

    state_t state, next_state;
    logic [7:0] timeout_cnt;    // FIX 4: Watchdog
    logic timeout;

    assign timeout = &timeout_cnt;  // all 1s = 255

    // FIX 4: Timeout counter
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            timeout_cnt <= '0;
        else if (state == IDLE)
            timeout_cnt <= '0;
        else if (!timeout)
            timeout_cnt <= timeout_cnt + 1;
    end

    // FIX 1: State register with async reset
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            state <= IDLE;
        else
            state <= next_state;
    end

    // Separated combinational logic
    always_comb begin
        next_state = state;
        ack        = 1'b0;
        busy       = 1'b0;
        error      = 1'b0;

        case (state)
            IDLE: begin
                if (req)
                    next_state = WAIT_DATA;
            end

            WAIT_DATA: begin
                busy = 1'b1;
                if (timeout) begin
                    next_state = IDLE;
                    error      = 1'b1;
                end else if (data_ready)
                    next_state = PROCESS;
            end

            PROCESS: begin
                busy = 1'b1;
                next_state = RESPOND;
            end

            RESPOND: begin
                ack  = 1'b1;
                busy = 1'b1;
                next_state = IDLE;
            end

            // FIX 2: Default recovery
            default: begin
                next_state = IDLE;
                error      = 1'b1;
            end
        endcase
    end
endmodule

Exercise 2: Write Assertions for a UART FSM

Given this UART transmitter FSM, write SystemVerilog assertions to verify safety properties.

이 UART 송신기 FSM이 주어졌을 때, 안전 속성을 검증하는 SystemVerilog 어설션을 작성하세요.

typedef enum logic [2:0] {
    TX_IDLE  = 3'b001,  // one-hot
    TX_START = 3'b010,
    TX_DATA  = 3'b100
} tx_state_t;
tx_state_t tx_state;

🎯 Write These Assertions

🎯 다음 어설션을 작성하세요

  1. State must always be one-hot
  2. 상태는 항상 원-핫이어야 함
  3. State must always be one of the three valid states
  4. 상태는 항상 세 가지 유효한 상태 중 하나여야 함
  5. From TX_DATA, must reach TX_IDLE within 20 cycles
  6. TX_DATA에서 20 사이클 이내에 TX_IDLE에 도달해야 함
  7. TX_START must last exactly 1 cycle
  8. TX_START는 정확히 1 사이클 지속해야 함
// 1. One-hot check
assert_onehot: assert property (
    @(posedge clk) disable iff (!rst_n)
    $onehot(tx_state)
) else $error("tx_state not one-hot: %b", tx_state);

// 2. Valid state check
assert_valid: assert property (
    @(posedge clk) disable iff (!rst_n)
    tx_state inside {TX_IDLE, TX_START, TX_DATA}
) else $error("Invalid state: %b", tx_state);

// 3. Liveness: TX_DATA → TX_IDLE within 20 cycles
assert_data_to_idle: assert property (
    @(posedge clk) disable iff (!rst_n)
    (tx_state == TX_DATA) |-> ##[1:20] (tx_state == TX_IDLE)
) else $error("TX_DATA stuck > 20 cycles");

// 4. TX_START lasts exactly 1 cycle
assert_start_one_cycle: assert property (
    @(posedge clk) disable iff (!rst_n)
    (tx_state == TX_START) |=> (tx_state != TX_START)
) else $error("TX_START lasted more than 1 cycle");

Exercise 3: Design a Safe SPI Controller FSM

Design a complete safe FSM for a simple SPI master controller with these states: IDLE → LOAD → SHIFT (8 bits) → DONE → IDLE. Include all safety features from today's lesson.

간단한 SPI 마스터 컨트롤러를 위한 완전한 안전 FSM을 설계하세요: IDLE → LOAD → SHIFT (8비트) → DONE → IDLE. 오늘 수업의 모든 안전 기능을 포함하세요.

🎯 Requirements

🎯 요구사항

  • Async assert, sync deassert reset
  • 비동기 어서트, 동기 디어서트 리셋
  • 3-bit counter to track shift progress
  • 시프트 진행을 추적하는 3비트 카운터
  • Timeout watchdog (64 cycles)
  • 타임아웃 워치독 (64 사이클)
  • Default case recovery + error flag
  • Default 케이스 복구 + 에러 플래그
  • At least 2 SVA assertions
  • 최소 2개의 SVA 어설션

💡 Hints

  • SHIFT state transitions to DONE when bit_count reaches 7
  • SHIFT 상태는 bit_count가 7에 도달하면 DONE으로 전이
  • Use the safe FSM template from the lesson as your starting point
  • 수업의 안전 FSM 템플릿을 시작점으로 사용하세요
  • Consider: what happens if the SPI slave doesn't respond?
  • 고려: SPI 슬레이브가 응답하지 않으면 어떻게 되나요?
module spi_master_safe (
    input  logic       clk,
    input  logic       rst_n,
    input  logic       start,
    input  logic [7:0] tx_data,
    output logic       sclk,
    output logic       mosi,
    output logic       cs_n,
    output logic       done,
    output logic       error
);

    typedef enum logic [2:0] {
        S_IDLE  = 3'b000,
        S_LOAD  = 3'b001,
        S_SHIFT = 3'b010,
        S_DONE  = 3'b100
    } state_t;

    state_t state, next_state;
    logic [7:0] shift_reg;
    logic [2:0] bit_cnt;
    logic [5:0] wd_cnt;     // watchdog (64 cycles)
    logic wd_timeout;

    assign wd_timeout = &wd_cnt;
    assign mosi = shift_reg[7];

    // ── Watchdog counter ──
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            wd_cnt <= '0;
        else if (state == S_IDLE)
            wd_cnt <= '0;
        else if (!wd_timeout)
            wd_cnt <= wd_cnt + 1;
    end

    // ── State register ──
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            state <= S_IDLE;
        else
            state <= next_state;
    end

    // ── Shift register & bit counter ──
    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            shift_reg <= '0;
            bit_cnt   <= '0;
        end else if (state == S_LOAD) begin
            shift_reg <= tx_data;
            bit_cnt   <= '0;
        end else if (state == S_SHIFT) begin
            shift_reg <= {shift_reg[6:0], 1'b0};
            bit_cnt   <= bit_cnt + 1;
        end
    end

    // ── Next-state logic ──
    always_comb begin
        next_state = state;
        cs_n       = 1'b1;
        sclk       = 1'b0;
        done       = 1'b0;
        error      = 1'b0;

        case (state)
            S_IDLE: begin
                if (start)
                    next_state = S_LOAD;
            end

            S_LOAD: begin
                cs_n = 1'b0;
                next_state = S_SHIFT;
            end

            S_SHIFT: begin
                cs_n = 1'b0;
                sclk = 1'b1;
                if (wd_timeout) begin
                    next_state = S_IDLE;
                    error      = 1'b1;
                end else if (bit_cnt == 3'd7)
                    next_state = S_DONE;
            end

            S_DONE: begin
                done = 1'b1;
                next_state = S_IDLE;
            end

            default: begin
                next_state = S_IDLE;
                error      = 1'b1;
            end
        endcase
    end

    // ── SVA Assertions ──
    // synopsys translate_off
    assert_valid_state: assert property (
        @(posedge clk) disable iff (!rst_n)
        state inside {S_IDLE, S_LOAD, S_SHIFT, S_DONE}
    ) else $error("Illegal SPI state: %b", state);

    assert_returns_idle: assert property (
        @(posedge clk) disable iff (!rst_n)
        (state == S_LOAD) |-> ##[1:20] (state == S_IDLE)
    ) else $error("SPI stuck after LOAD");
    // synopsys translate_on

endmodule
