4-bit Priority Encoder 4비트 우선순위 인코더 Easy

📋 Your Task

📋 과제

Implement a 4-bit priority encoder using always_comb. The encoder should output the index of the highest-priority active request bit.

always_comb를 사용하여 4비트 우선순위 인코더를 구현하세요. 인코더는 가장 높은 우선순위의 활성 요청 비트 인덱스를 출력해야 합니다.

  • Input: req[3:0] — 4 request lines (bit 3 = highest priority)
  • 입력: req[3:0] — 4개의 요청 라인 (비트 3 = 최고 우선순위)
  • Output: grant[1:0] — index of highest active bit
  • 출력: grant[1:0] — 최고 활성 비트의 인덱스
  • Output: valid — 1 if any request is active, 0 otherwise
  • 출력: valid — 요청이 있으면 1, 없으면 0

📝 Starter Code

📝 시작 코드

// Priority Encoder - Fill in the always_comb block
module priority_encoder (
    input  logic [3:0] req,
    output logic [1:0] grant,
    output logic        valid
);

    always_comb begin
        // TODO: Implement priority encoding
        // Hint: Check from highest priority (req[3]) to lowest (req[0])
        // Remember: Assign ALL outputs in ALL paths!
        
        valid = 1'b0;    // Default
        grant = 2'd0;    // Default
        
        // Your code here...
        
    end

endmodule

// ============ TESTBENCH ============
module tb_priority_encoder;
    logic [3:0] req;
    logic [1:0] grant;
    logic        valid;
    
    priority_encoder dut (.*);
    
    initial begin
        $display("Testing Priority Encoder");
        $display("========================");
        
        // Test: No requests
        req = 4'b0000; #10;
        $display("req=%b → grant=%d, valid=%b (expect: grant=0, valid=0)", req, grant, valid);
        
        // Test: Only req[0]
        req = 4'b0001; #10;
        $display("req=%b → grant=%d, valid=%b (expect: grant=0, valid=1)", req, grant, valid);
        
        // Test: req[2] highest
        req = 4'b0101; #10;
        $display("req=%b → grant=%d, valid=%b (expect: grant=2, valid=1)", req, grant, valid);
        
        // Test: req[3] highest (all active)
        req = 4'b1111; #10;
        $display("req=%b → grant=%d, valid=%b (expect: grant=3, valid=1)", req, grant, valid);
        
        // Test: req[1] only
        req = 4'b0010; #10;
        $display("req=%b → grant=%d, valid=%b (expect: grant=1, valid=1)", req, grant, valid);
        
        $display("========================");
        $display("Test complete!");
        $finish;
    end
endmodule

8-bit Register with Enable 인에이블이 있는 8비트 레지스터 Easy

📋 Your Task

📋 과제

Implement an 8-bit register with asynchronous reset and synchronous enable using always_ff.

always_ff를 사용하여 비동기 리셋과 동기 인에이블이 있는 8비트 레지스터를 구현하세요.

  • Input: clk, rst_n (active low), en, d[7:0]
  • 입력: clk, rst_n (액티브 로우), en, d[7:0]
  • Output: q[7:0]
  • 출력: q[7:0]
  • Behavior: Reset to 0, capture d when en=1 on rising clock edge
  • 동작: 0으로 리셋, 상승 클럭 에지에서 en=1일 때 d를 캡처

📝 Starter Code

📝 시작 코드

// 8-bit Register with Enable - Fill in the always_ff block
module register_en (
    input  logic        clk,
    input  logic        rst_n,   // Active-low async reset
    input  logic        en,      // Clock enable
    input  logic [7:0] d,
    output logic [7:0] q
);

    // TODO: Implement the register
    // Requirements:
    // 1. Asynchronous reset (rst_n in sensitivity list)
    // 2. When rst_n=0, output q should be 0
    // 3. When en=1 and rising clock, capture d
    // 4. When en=0, hold current value
    
    always_ff @(/* sensitivity list here */) begin
        // Your code here...
    end

endmodule

// ============ TESTBENCH ============
module tb_register_en;
    logic        clk, rst_n, en;
    logic [7:0] d, q;
    
    register_en dut (.*);
    
    // Clock generation: 10ns period
    always #5 clk = ~clk;
    
    initial begin
        $display("Testing 8-bit Register with Enable");
        $display("====================================");
        
        // Initialize
        clk = 0; rst_n = 0; en = 0; d = 8'hAA;
        
        // Test reset
        #12;
        $display("After reset: q=%h (expect: 00)", q);
        
        // Release reset
        rst_n = 1;
        
        // Test: en=0, data should NOT load
        d = 8'h55; en = 0;
        @(posedge clk); #1;
        $display("en=0, d=55: q=%h (expect: 00, no load)", q);
        
        // Test: en=1, data should load
        d = 8'hBE; en = 1;
        @(posedge clk); #1;
        $display("en=1, d=BE: q=%h (expect: BE)", q);
        
        // Test: en=1, new data
        d = 8'hEF;
        @(posedge clk); #1;
        $display("en=1, d=EF: q=%h (expect: EF)", q);
        
        // Test: en=0, should hold
        d = 8'h00; en = 0;
        @(posedge clk); #1;
        $display("en=0, d=00: q=%h (expect: EF, hold)", q);
        
        // Test: async reset while running
        rst_n = 0; #2;
        $display("Async reset: q=%h (expect: 00)", q);
        
        $display("====================================");
        $display("Test complete!");
        $finish;
    end
endmodule

The sensitivity list for async reset needs both clock and reset:

always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n)
        q <= 8'h00;

        // ...
end

ALU with Registered Output 레지스터 출력이 있는 ALU Medium

📋 Your Task

📋 과제

Build an 8-bit ALU that combines always_comb for the operation logic and always_ff for registering the output. This is a common design pattern!

연산 논리에는 always_comb를, 출력 레지스터에는 always_ff를 결합한 8비트 ALU를 만드세요. 이것은 일반적인 설계 패턴입니다!

  • Operations: ADD (00), SUB (01), AND (10), OR (11)
  • 연산: ADD (00), SUB (01), AND (10), OR (11)
  • Combinational: Compute result based on opcode
  • 조합: opcode에 따라 결과 계산
  • Sequential: Register the result on clock edge
  • 순차: 클럭 에지에서 결과 레지스터

📝 Starter Code

📝 시작 코드

// ALU with Registered Output
module alu_registered (
    input  logic        clk,
    input  logic        rst_n,
    input  logic [7:0] a,
    input  logic [7:0] b,
    input  logic [1:0] op,       // 00=ADD, 01=SUB, 10=AND, 11=OR
    input  logic        valid_in, // Input valid signal
    output logic [7:0] result,
    output logic        valid_out // Registered valid
);

    // Internal signal for combinational result
    logic [7:0] alu_out;

    // TODO 1: Combinational ALU logic
    always_comb begin
        case (op)
            // Implement all 4 operations
            // Don't forget default!
            default: alu_out = 8'h00;
        endcase
    end

    // TODO 2: Sequential output register
    always_ff @(/* sensitivity list */) begin
        // Reset result and valid_out
        // On clock: register alu_out → result
        // On clock: register valid_in → valid_out
    end

endmodule

// ============ TESTBENCH ============
module tb_alu_registered;
    logic        clk, rst_n, valid_in, valid_out;
    logic [7:0] a, b, result;
    logic [1:0] op;
    
    alu_registered dut (.*);
    
    always #5 clk = ~clk;
    
    initial begin
        $display("Testing ALU with Registered Output");
        $display("====================================");
        
        clk = 0; rst_n = 0; valid_in = 0;
        a = 0; b = 0; op = 0;
        
        #12; rst_n = 1;
        
        // Test ADD: 10 + 5 = 15
        a = 8'd10; b = 8'd5; op = 2'b00; valid_in = 1;
        @(posedge clk); #1;
        $display("ADD: %d + %d = %d, valid=%b (expect: 15, 1)", 10, 5, result, valid_out);
        
        // Test SUB: 20 - 8 = 12
        a = 8'd20; b = 8'd8; op = 2'b01;
        @(posedge clk); #1;
        $display("SUB: %d - %d = %d, valid=%b (expect: 12, 1)", 20, 8, result, valid_out);
        
        // Test AND: 0xF0 & 0x0F = 0x00
        a = 8'hF0; b = 8'h0F; op = 2'b10;
        @(posedge clk); #1;
        $display("AND: F0 & 0F = %h, valid=%b (expect: 00, 1)", result, valid_out);
        
        // Test OR: 0xF0 | 0x0F = 0xFF
        a = 8'hF0; b = 8'h0F; op = 2'b11;
        @(posedge clk); #1;
        $display("OR:  F0 | 0F = %h, valid=%b (expect: FF, 1)", result, valid_out);
        
        // Test valid_out follows valid_in
        valid_in = 0;
        @(posedge clk); #1;
        $display("Valid pipeline: valid_out=%b (expect: 0)", valid_out);
        
        $display("====================================");
        $display("Test complete!");
        $finish;
    end
endmodule

Structure your code like this:

코드를 다음과 같이 구조화하세요:

// Combinational
always_comb begin
    case (op)
        2'b00: alu_out = a + b;
        2'b01: alu_out = a - b;
        // ...
    endcase
end

// Sequential
always_ff @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        result    <= 8'd0;
        valid_out <= 1'b0;
    end else begin
        result    <= alu_out;
        valid_out <= valid_in;
    end
end
