import re, json, pathlib

def extract_verilog_modules(filepath: str) -> list[dict]:
    """Extract individual modules from a Verilog file."""
    text = pathlib.Path(filepath).read_text()
    # Match module ... endmodule blocks
    pattern = r'(module\s+\w+[\s\S]*?endmodule)'
    modules = re.findall(pattern, text)
    results = []
    for mod in modules:
        name_match = re.match(r'module\s+(\w+)', mod)
        results.append({
            "name": name_match.group(1) if name_match else "unknown",
            "code": mod,
            "file": filepath
        })
    return results

import re

def parse_sim_log(log_path: str) -> list[dict]:
    """Extract errors, warnings, and test results from verification logs."""
    entries = []
    with open(log_path) as f:
        for line in f:
            # UVM errors
            if re.search(r'UVM_ERROR|UVM_FATAL', line):
                entries.append({"type": "error", "line": line.strip()})
            # Assertion failures
            elif re.search(r'Assertion.*failed|assert.*error', line, re.I):
                entries.append({"type": "assertion", "line": line.strip()})
            # Timing violations
            elif re.search(r'timing violation|setup.*violation|hold.*violation', line, re.I):
                entries.append({"type": "timing", "line": line.strip()})
            # Test pass/fail
            elif re.search(r'TEST (PASSED|FAILED)', line):
                entries.append({"type": "test_result", "line": line.strip()})
    return entries

def generate_training_pairs(chunk: str, source_file: str) -> list[dict]:

Quality Filtering품질 필터링

def quality_score(pair: dict) -> float:
    """Score a Q&A pair from 0-1 based on quality heuristics."""
    score = 0.0
    q, a = pair["q"], pair["a"]

    # Length checks
    if len(a) > 100: score += 0.2   # Answers should be substantial
    if len(a) > 300: score += 0.1   # Bonus for detailed answers
    if len(q) > 20: score += 0.1    # Questions shouldn't be too short

    # Technical content indicators
    if any(kw in a.lower() for kw in ['register', 'clock', 'timing', 'voltage',
        'protocol', 'signal', 'module', 'frequency', 'latency']):
        score += 0.2

    # Has code or specific values
    if '```' in a or re.search(r'\d+\s*(MHz|ns|mV|Gbps|dB)', a):
        score += 0.2

    # Not too repetitive
    if q.lower() not in a.lower()[:50]:
        score += 0.1

    # Penalize very short or generic answers
    if "I don't know" in a or len(a) < 50:
        score = 0.0

    return min(score, 1.0)

# Filter: keep only pairs scoring > 0.5

Deduplication with MinHashMinHash를 이용한 중복 제거

from datasketch import MinHash, MinHashLSH

def deduplicate_pairs(pairs: list[dict], threshold: float = 0.8) -> list[dict]:
    """Remove near-duplicate Q&A pairs using MinHash LSH."""
    lsh = MinHashLSH(threshold=threshold, num_perm=128)
    unique_pairs = []

    for i, pair in enumerate(pairs):
        m = MinHash(num_perm=128)
        for word in pair["q"].lower().split():
            m.update(word.encode('utf8'))

        key = f"pair_{i}"
        if not lsh.query(m):  # No similar pair found
            lsh.insert(key, m)
            unique_pairs.append(pair)

    print(f"Dedup: {len(pairs)} → {len(unique_pairs)} ({len(pairs)-len(unique_pairs)} removed)")
    return unique_pairs

from fastapi import FastAPI, Header, HTTPException
from fastapi.middleware.cors import CORSMiddleware
import httpx, time, json

app = FastAPI(title="Digital Design AI API")
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

OLLAMA_URL = "http://localhost:11434"
API_KEYS = {"minjun-key-123", "team-key-456"}  # Simple API key auth

@app.post("/v1/chat/completions")
async def chat(request: dict, authorization: str = Header(None)):
    # Auth check
    token = (authorization or "").replace("Bearer ", "")
    if token not in API_KEYS:
        raise HTTPException(401, "Invalid API key")

    # Log the request
    print(f"[{time.strftime('%H:%M:%S')}] User query: {request['messages'][-1]['content'][:80]}...")

    # Forward to Ollama
    async with httpx.AsyncClient(timeout=120) as client:
        resp = await client.post(
            f"{OLLAMA_URL}/v1/chat/completions",
            json={**request, "model": "design-ai"}
        )
    return resp.json()

# Run: uvicorn api_gateway:app --host 0.0.0.0 --port 8000
